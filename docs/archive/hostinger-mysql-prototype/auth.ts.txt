// ARCHIVED PROTOTYPE: not imported by the production application.
import "server-only";

import { createHash, randomBytes } from "node:crypto";
import { compare } from "bcryptjs";
import type { RowDataPacket } from "mysql2/promise";
import { cookies } from "next/headers";
import type { DashboardUserRole } from "@/types";
import { getHostingerPool, mysqlRows, withMysqlTransaction } from "./db";

const SESSION_COOKIE = "bs-hostinger-session";
const SESSION_TTL_SECONDS = 60 * 60 * 24 * 30;

type AuthRow = RowDataPacket & {
  user_id: string;
  email: string;
  display_name: string;
  password_hash: string;
  role: DashboardUserRole;
  client_id: string | null;
  client_name: string | null;
};

type SessionRow = RowDataPacket & {
  user_id: string;
  email: string;
  display_name: string;
  role: DashboardUserRole;
  client_id: string | null;
  client_name: string | null;
};

export type HostingerPortalActor = {
  userId: string;
  email: string;
  name: string;
  role: DashboardUserRole;
  clientId: string | null;
  clientName: string | null;
};

const digest = (value: string) => createHash("sha256").update(value).digest("hex");

export async function authenticateHostingerUser(email: string, password: string) {
  const normalizedEmail = email.trim().toLowerCase();
  const rows = await mysqlRows<AuthRow[]>(
    `SELECT
       u.id AS user_id,
       u.email,
       u.display_name,
       u.password_hash,
       m.role,
       c.slug AS client_id,
       c.name AS client_name
     FROM portal_users u
     INNER JOIN portal_memberships m ON m.user_id = u.id AND m.status = 'active'
     LEFT JOIN portal_clients c ON c.id = m.client_id
     WHERE u.email = ? AND u.status = 'active'
     ORDER BY m.created_at ASC
     LIMIT 1`,
    [normalizedEmail],
  );
  const row = rows[0];
  if (!row || !(await compare(password, row.password_hash))) return null;
  return toActor(row);
}

export async function createHostingerSession(actor: HostingerPortalActor) {
  const rawToken = randomBytes(36).toString("base64url");
  const tokenHash = digest(rawToken);
  const expiresAt = new Date(Date.now() + SESSION_TTL_SECONDS * 1000);

  await withMysqlTransaction(async connection => {
    await connection.execute(
      "DELETE FROM portal_sessions WHERE user_id = ? AND expires_at <= UTC_TIMESTAMP()",
      [actor.userId],
    );
    await connection.execute(
      `INSERT INTO portal_sessions
        (id, user_id, token_hash, expires_at, created_at, last_seen_at)
       VALUES (UUID(), ?, ?, ?, UTC_TIMESTAMP(), UTC_TIMESTAMP())`,
      [actor.userId, tokenHash, expiresAt],
    );
  });

  const cookieStore = await cookies();
  cookieStore.set(SESSION_COOKIE, rawToken, {
    httpOnly: true,
    sameSite: "lax",
    secure: process.env.NODE_ENV === "production",
    path: "/",
    maxAge: SESSION_TTL_SECONDS,
  });
}

export async function readHostingerSession(): Promise<HostingerPortalActor | null> {
  const cookieStore = await cookies();
  const rawToken = cookieStore.get(SESSION_COOKIE)?.value;
  if (!rawToken) return null;

  const rows = await mysqlRows<SessionRow[]>(
    `SELECT
       u.id AS user_id,
       u.email,
       u.display_name,
       m.role,
       c.slug AS client_id,
       c.name AS client_name
     FROM portal_sessions s
     INNER JOIN portal_users u ON u.id = s.user_id AND u.status = 'active'
     INNER JOIN portal_memberships m ON m.user_id = u.id AND m.status = 'active'
     LEFT JOIN portal_clients c ON c.id = m.client_id
     WHERE s.token_hash = ? AND s.expires_at > UTC_TIMESTAMP()
     ORDER BY m.created_at ASC
     LIMIT 1`,
    [digest(rawToken)],
  );
  return rows[0] ? toActor(rows[0]) : null;
}

export async function deleteHostingerSession() {
  const cookieStore = await cookies();
  const rawToken = cookieStore.get(SESSION_COOKIE)?.value;
  if (rawToken) {
    await getHostingerPool().execute(
      "DELETE FROM portal_sessions WHERE token_hash = ?",
      [digest(rawToken)],
    );
  }
  cookieStore.delete(SESSION_COOKIE);
}

function toActor(row: AuthRow | SessionRow): HostingerPortalActor {
  return {
    userId: row.user_id,
    email: row.email,
    name: row.display_name,
    role: row.role,
    clientId: row.client_id,
    clientName: row.client_name,
  };
}
